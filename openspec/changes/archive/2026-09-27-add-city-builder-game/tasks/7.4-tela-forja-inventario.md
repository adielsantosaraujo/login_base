## Notas de implementação

### `ForjaView.vue` — tela com formulário e inventário
```vue
<template>
  <div class="forja-view">
    <h1>Forja ({{ nivelForja }})</h1>
    
    <Card class="formulario">
      <template #title>Nova ordem</template>
      
      <div class="grid">
        <div>
          <label>Modelo</label>
          <Select v-model="modelo" :options="modelos" optionLabel="nome" optionValue="id" />
        </div>
        <div>
          <label>Nível</label>
          <InputNumber v-model="nivel" :min="1" :max="5" />
        </div>
        <div>
          <label>Quantidade</label>
          <InputNumber v-model="quantidade" :min="1" :max="5" />
        </div>
      </div>
      
      <div class="custo">
        <p>Custo:</p>
        <div v-for="[recurso, valor] in custoCalculado" :key="recurso">
          {{ recurso }}: {{ valor }}
        </div>
      </div>
      
      <Button
        label="Forjar"
        @click="forjar"
        :disabled="!podeForjar || temOrdemEmAndamento"
      />
      
      <div v-if="temOrdemEmAndamento" class="ordem-em-andamento">
        <ProgressBar :value="percentualOrdem" />
        <p>Ordem em andamento</p>
      </div>
    </Card>
    
    <h2>Inventário</h2>
    <DataTable
      :value="vila?.itens ?? []"
      paginator
      :rows="10"
      responsiveLayout="scroll"
    >
      <Column field="modelo" header="Modelo" />
      <Column field="categoria" header="Categoria" />
      <Column field="nivel" header="Nível" />
      <Column field="ataque" header="Ataque" />
      <Column field="defesa" header="Defesa" />
      <Column field="status" header="Status" />
      <Column field="origem" header="Origem" />
    </DataTable>
  </div>
</template>

<script setup lang="ts">
import { computed, ref, onMounted } from 'vue'
import { useVila } from '../composables/useVila'
import { jogoAPI } from '../api/jogo'
import { useToast } from 'primevue/usetoast'
import type { ModeloItem } from '../api/tipos'

const { vila, catalogo, carregar, iniciarPoll } = useVila()
const toast = useToast()

const modelo = ref<string>('ESPADA')
const nivel = ref<number>(1)
const quantidade = ref<number>(1)

const nivelForja = computed(() => {
  return vila.value?.predios.find(p => p.tipo === 'FORJA')?.nivel ?? 0
})

const modelos = computed(() => {
  return catalogo.value?.modelos ?? []
})

const custoCalculado = computed(() => {
  const m = catalogo.value?.modelos.find(x => x.id === modelo.value)
  if (!m) return {}
  
  const base = m.receita
  const totalCusto: Record<string, number> = {}
  
  for (const [recurso, valor] of Object.entries(base)) {
    totalCusto[recurso] = (valor as number) * nivel.value * quantidade.value
  }
  
  return totalCusto
})

const podeForjar = computed(() => {
  if (nivelForja.value === 0) return false
  if (nivel.value > nivelForja.value) return false
  
  // Validar recursos
  const recursos = vila.value?.recursos ?? {}
  for (const [recurso, custo] of Object.entries(custoCalculado.value)) {
    if ((recursos[recurso as keyof typeof recursos] ?? 0) < (custo as number)) {
      return false
    }
  }
  
  return true
})

const temOrdemEmAndamento = computed(() => {
  return vila.value?.ordens.some(o => o.categoria === 'FORJA') ?? false
})

const percentualOrdem = computed(() => {
  const ordem = vila.value?.ordens.find(o => o.categoria === 'FORJA')
  if (!ordem) return 0
  const inicio = new Date(ordem.iniciadaEm).getTime()
  const conclusao = new Date(ordem.concluiEm).getTime()
  const agora = new Date(vila.value?.agora ?? Date.now()).getTime()
  return Math.min(100, ((agora - inicio) / (conclusao - inicio)) * 100)
})

async function forjar() {
  try {
    await jogoAPI.forjar(modelo.value, nivel.value, quantidade.value)
    await carregar()
  } catch (e: any) {
    toast.add({ severity: 'error', detail: e.mensagem, life: 3000 })
  }
}

onMounted(() => {
  carregar()
  iniciarPoll()
})
</script>

<style scoped>
.forja-view {
  padding: 16px;
}

.formulario {
  margin-bottom: 24px;
  max-width: 600px;
}

.grid {
  display: grid;
  grid-template-columns: 1fr 1fr 1fr;
  gap: 16px;
  margin-bottom: 16px;
}

.custo {
  background: var(--surface);
  padding: 16px;
  margin: 16px 0;
  border-radius: 4px;
}
</style>
```

## Critérios de aceite
- `npm run build` sem erros.
- ForjaView renderiza Select de modelo/nível/quantidade.
- Custo calculado corretamente: `custo_base × nível × quantidade`.
- Botão Forjar desabilitado se forja N0, nível > forja, ou sem recursos.
- DataTable exibe inventário com filtro possível.
- Ordem em andamento mostra ProgressBar.
- Erros 422 exibem Toast.
