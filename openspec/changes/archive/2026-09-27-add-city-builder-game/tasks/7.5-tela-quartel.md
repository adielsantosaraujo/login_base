# Task 7.5 — Tela do quartel

## Objetivo

## Notas de implementação

### `QuartelView.vue` — tela com seleção de tropas e inventário
```vue
<template>
  <div class="quartel-view">
    <h1>Quartel ({{ nivelQuartel }})</h1>
    
    <Card class="formulario">
      <template #title>Treinar unidade</template>
      
      <div class="grid">
        <div>
          <label>Tipo</label>
          <Select v-model="tipo" :options="tiposDisponiveis" optionLabel="nome" optionValue="id" />
        </div>
        <div v-if="tipo">
          <label>Arma ({{ armaExigida }})</label>
          <Select
            v-model="armaId"
            :options="armasDisponiveis"
            optionLabel="descricao"
            optionValue="id"
          />
        </div>
        <div>
          <label>Armadura</label>
          <Select
            v-model="armaduraId"
            :options="armadurasDisponiveis"
            optionLabel="descricao"
            optionValue="id"
          />
        </div>
      </div>
      
      <Button
        label="Treinar"
        @click="treinar"
        :disabled="!podeTreinar || temOrdemEmAndamento"
      />
      
      <div v-if="temOrdemEmAndamento" class="ordem-em-andamento">
        <ProgressBar :value="percentualOrdem" />
        <p>Treino em andamento</p>
      </div>
    </Card>
    
    <div class="capacidade">
      <h2>Capacidade do exército: {{ unidadesAtivas }}/{{ capacidadeTotal }}</h2>
    </div>
    
    <h2>Unidades</h2>
    <DataTable
      :value="vila?.unidades ?? []"
      paginator
      :rows="10"
      responsiveLayout="scroll"
    >
      <Column field="tipo" header="Tipo" />
      <Column field="hp" header="HP" />
      <Column field="ataque" header="Ataque" />
      <Column field="defesa" header="Defesa" />
      <Column field="alcance" header="Alcance" />
      <Column field="movimento" header="Movimento" />
      <Column field="status" header="Status" />
    </DataTable>
  </div>
</template>

<script setup lang="ts">
import { computed, ref, onMounted } from 'vue'
import { useVila } from '../composables/useVila'
import { jogoAPI } from '../api/jogo'
import { useToast } from 'primevue/usetoast'

const { vila, catalogo, carregar, iniciarPoll } = useVila()
const toast = useToast()

const tipo = ref<string>('SOLDADO')
const armaId = ref<number | null>(null)
const armaduraId = ref<number | null>(null)

const nivelQuartel = computed(() => {
  return vila.value?.predios.find(p => p.tipo === 'QUARTEL')?.nivel ?? 0
})

const tiposDisponiveis = computed(() => {
  return (catalogo.value?.tropas ?? []).filter(t => {
    if (t.quartelMinimo > nivelQuartel.value) return false
    return true
  })
})

const armaExigida = computed(() => {
  const t = catalogo.value?.tropas.find(x => x.id === tipo.value)
  return t?.armaExigida ?? '—'
})

const armasDisponiveis = computed(() => {
  const t = catalogo.value?.tropas.find(x => x.id === tipo.value)
  if (!t) return []
  
  return (vila.value?.itens ?? []).filter(item => {
    if (item.categoria !== 'ARMA') return false
    if (item.status !== 'DISPONIVEL') return false
    if (item.modelo !== t.armaExigida) return false
    return true
  }).map(item => ({
    id: item.id,
    descricao: `${item.modelo} N${item.nivel}`
  }))
})

const armadurasDisponiveis = computed(() => {
  return (vila.value?.itens ?? []).filter(item => {
    if (item.categoria !== 'ARMADURA') return false
    if (item.status !== 'DISPONIVEL') return false
    return true
  }).map(item => ({
    id: item.id,
    descricao: `${item.modelo} N${item.nivel}`
  }))
})

const podeTreinar = computed(() => {
  if (nivelQuartel.value === 0) return false
  if (!armaId.value || !armaduraId.value) return false
  
  // Validar comida
  const comida = vila.value?.recursos.COMIDA ?? 0
  const t = catalogo.value?.tropas.find(x => x.id === tipo.value)
  if ((t?.comida ?? 0) > comida) return false
  
  return true
})

const temOrdemEmAndamento = computed(() => {
  return vila.value?.ordens.some(o => o.categoria === 'TREINO') ?? false
})

const unidadesAtivas = computed(() => {
  const unidades = (vila.value?.unidades ?? []).length
  const orden = temOrdemEmAndamento.value ? 1 : 0
  return unidades + orden
})

const capacidadeTotal = computed(() => {
  return vila.value?.capacidadeExercito ?? 0
})

const percentualOrdem = computed(() => {
  const ordem = vila.value?.ordens.find(o => o.categoria === 'TREINO')
  if (!ordem) return 0
  const inicio = new Date(ordem.iniciadaEm).getTime()
  const conclusao = new Date(ordem.concluiEm).getTime()
  const agora = new Date(vila.value?.agora ?? Date.now()).getTime()
  return Math.min(100, ((agora - inicio) / (conclusao - inicio)) * 100)
})

async function treinar() {
  if (!armaId.value || !armaduraId.value) return
  
  try {
    await jogoAPI.treinar(tipo.value, armaId.value, armaduraId.value)
    await carregar()
  } catch (e: any) {
    toast.add({ severity: 'error', detail: e.mensagem, life: 3000 })
  }
}

onMounted(() => {
  carregar()
  iniciarPoll()
})
</script>

<style scoped>
.quartel-view {
  padding: 16px;
}

.formulario {
  margin-bottom: 24px;
  max-width: 600px;
}

.grid {
  display: grid;
  grid-template-columns: 1fr 1fr 1fr;
  gap: 16px;
  margin-bottom: 16px;
}

.capacidade {
  background: var(--surface);
  padding: 16px;
  margin: 16px 0;
  border-radius: 4px;
}
</style>
```

## Critérios de aceite
- `npm run build` sem erros.
- QuartelView renderiza Select de tipo/arma/armadura.
- Armas filtram por tipo exigido; armaduras filtram por disponibilidade.
- Botão Treinar desabilitado se quartel N0, itens não selecionados, ou capacidade cheia.
- Capacidade do exército exibe corretamente (unidades + ordem em treino).
- DataTable exibe unidades com atributos.
- Erros 422 exibem Toast.
