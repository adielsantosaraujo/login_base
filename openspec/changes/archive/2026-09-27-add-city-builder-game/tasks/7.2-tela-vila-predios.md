### `CartaoPredio.vue` — card de prédio com upgrade
```vue
<template>
  <Card class="cartao-predio">
    <template #header>
      <h3>{{ predio.tipo }}</h3>
      <span>N{{ predio.nivel }}</span>
    </template>
    <div>
      <p v-if="predio.nivel === predio.nivelMaximo">Nível máximo</p>
      <div v-else-if="predio.proximoNivel">
        <p>Próximo: N{{ predio.proximoNivel.nivel }}</p>
        <p>Custo:</p>
        <div v-for="[recurso, valor] in Object.entries(predio.proximoNivel.custo)" :key="recurso">
          {{ recurso }}: {{ valor }}
        </div>
        <p>Tempo: {{ predio.proximoNivel.tempoSegundos }}s</p>
        
        <ProgressBar v-if="ordemEmAndamento" :value="percentualConclusao" />
        
        <Button
          label="Melhorar"
          @click="melhorar"
          :disabled="!podemelhorar || ordemEmAndamento"
          :severity="bloqueioSeveridade"
        />
        <small v-if="predio.proximoNivel.bloqueio">
          Bloqueado: {{ predio.proximoNivel.bloqueio }}
        </small>
      </div>
    </div>
  </Card>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import { useVila } from '../composables/useVila'
import { jogoAPI } from '../api/jogo'
import { useToast } from 'primevue/usetoast'
import type { PredioDto, OrdemDto } from '../api/tipos'

const props = defineProps<{ predio: PredioDto }>()
const { vila, carregar } = useVila()
const toast = useToast()

const ordem = computed(() => {
  return vila.value?.ordens.find(o => o.categoria === 'CONSTRUCAO' && o.alvo === props.predio.tipo)
})

const ordemEmAndamento = computed(() => !!ordem.value)

const percentualConclusao = computed(() => {
  if (!ordem.value) return 0
  const inicio = new Date(ordem.value.iniciadaEm).getTime()
  const conclusao = new Date(ordem.value.concluiEm).getTime()
  const agora = new Date(vila.value?.agora ?? Date.now()).getTime()
  return Math.min(100, ((agora - inicio) / (conclusao - inicio)) * 100)
})

const podemelhorar = computed(() => {
  if (!props.predio.proximoNivel) return false
  if (!props.predio.proximoNivel.bloqueio) return true
  return false  // bloqueio impede
})

const bloqueioSeveridade = computed(() => {
  return props.predio.proximoNivel?.bloqueio ? 'danger' : 'primary'
})

async function melhorar() {
  try {
    await jogoAPI.melhorar(props.predio.tipo)
    await carregar()
  } catch (e: any) {
    toast.add({ severity: 'error', detail: e.mensagem, life: 3000 })
  }
}
</script>

<style scoped>
.cartao-predio {
  min-width: 250px;
}
</style>
```

### `VilaView.vue` — grid de prédios
```vue
<template>
  <div class="vila-view">
    <h1>{{ vila?.nome ?? 'Vila' }}</h1>
    
    <h2>Prédios</h2>
    <div class="grid-predios">
      <CartaoPredio v-for="pred in vila?.predios" :key="pred.tipo" :predio="pred" />
    </div>
    
    <h2>Ordens</h2>
    <div v-if="vila?.ordens.length">
      <div v-for="ordem in vila.ordens" :key="ordem.id" class="ordem">
        <p>{{ ordem.categoria }} {{ ordem.alvo }} N{{ ordem.nivel }}</p>
        <p>Conclusão: {{ ordem.concluiEm }}</p>
      </div>
    </div>
    <p v-else>Nenhuma ordem em andamento</p>
  </div>
</template>

<script setup lang="ts">
import { onMounted } from 'vue'
import { useVila } from '../composables/useVila'
import CartaoPredio from '../components/CartaoPredio.vue'

const { vila, carregar, iniciarPoll } = useVila()

onMounted(() => {
  carregar()
  iniciarPoll()
})
</script>
```
