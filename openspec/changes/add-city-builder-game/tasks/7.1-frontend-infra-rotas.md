### `src/api/http.ts` — cliente HTTP com CSRF
```typescript
export async function request<T>(
  path: string,
  init?: RequestInit
): Promise<T> {
  const headers = new Headers(init?.headers)
  
  // Lê XSRF-TOKEN do cookie
  const token = document.cookie
    .split('; ')
    .find(row => row.startsWith('XSRF-TOKEN='))
    ?.split('=')[1]
  
  if (token && init?.method && ['POST', 'PUT', 'DELETE'].includes(init.method)) {
    headers.set('X-XSRF-TOKEN', token)
  }
  
  const resp = await fetch(path, {
    credentials: 'same-origin',
    ...init,
    headers
  })
  
  // 401 → redireciona para login
  if (resp.status === 401) {
    window.location.href = '/login'
    throw new Error('Unauthorized')
  }
  
  // Erro 4xx/5xx com corpo JSON
  if (!resp.ok) {
    const erro = await resp.json().catch(() => ({}))
    throw new ErroApi(erro.codigo, erro.mensagem, resp.status)
  }
  
  return resp.json()
}

export class ErroApi extends Error {
  constructor(public codigo: string, public mensagem: string, public status: number) {
    super(mensagem)
  }
}
```

### `src/composables/useVila.ts` — estado reativo singleton
```typescript
import { ref, computed } from 'vue'
import { jogoAPI } from '../api/jogo'
import type { VilaDto, CatalogoDto } from '../api/tipos'

const vila = ref<VilaDto | null>(null)
const catalogo = ref<CatalogoDto | null>(null)
let pollInterval: number | null = null

export function useVila() {
  async function carregar() {
    try {
      vila.value = await jogoAPI.consultarVila()
      catalogo.value ??= await jogoAPI.catalogo()
    } catch (e) {
      console.error('Erro ao carregar vila:', e)
    }
  }
  
  function iniciarPoll() {
    if (pollInterval) return
    pollInterval = window.setInterval(carregar, 5000)
  }
  
  function pararPoll() {
    if (pollInterval) {
      clearInterval(pollInterval)
      pollInterval = null
    }
  }
  
  const deslocamentoRelogio = computed(() => {
    if (!vila.value) return 0
    return new Date(vila.value.agora).getTime() - Date.now()
  })
  
  return { vila, catalogo, carregar, iniciarPoll, pararPoll, deslocamentoRelogio }
}
```

      form.method = 'POST'

  initializar()
  iniciarPoll()
})
