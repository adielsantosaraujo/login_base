### `GradeBatalha.vue` — componente de grade 8×8
```vue
<template>
  <div class="grade-batalha">
    <div class="grade" ref="gradeEl">
      <div
        v-for="(pos, idx) in todasAsPosicoes"
        :key="idx"
        class="celula"
        :class="{
          obstaculo: isObstaculo(pos),
          combatente: getCombatente(pos),
          selecionado: selecionado?.x === pos.x && selecionado?.y === pos.y,
          destino: isDestino(pos),
          alvo: isAlvo(pos),
        }"
        @click="clicar(pos)"
      >
        <span v-if="getCombatente(pos)" class="label">
          {{ getCombatente(pos)?.id }}
        </span>
      </div>
    </div>
    
    <div class="painel-selecionado">
      <div v-if="combatenteSelecionado">
        <h3>{{ combatenteSelecionado.id }}</h3>
        <p>{{ combatenteSelecionado.tipo }}</p>
        <p>HP: {{ combatenteSelecionado.hp }}/{{ combatenteSelecionado.hpMax }}</p>
        <p>Ataque: {{ combatenteSelecionado.ataque }} · Alcance: {{ combatenteSelecionado.alcance }}</p>
        <p>Defesa: {{ combatenteSelecionado.defesa }} · Movimento: {{ combatenteSelecionado.movimento }}</p>
        <p v-if="combatenteSelecionado.lado === 'JOGADOR'">Moveu: {{ combatenteSelecionado.moveu }}</p>
        <p v-if="combatenteSelecionado.lado === 'JOGADOR'">Agiu: {{ combatenteSelecionado.agiu }}</p>
      </div>
    </div>
    
    <div class="botoes-acao">
      <Button
        v-if="combatenteSelecionado?.lado === 'JOGADOR' && !combatenteSelecionado.agiu"
        label="Mover"
        @click="modoMover = true"
        :disabled="combatenteSelecionado.moveu"
      />
      <Button
        v-if="combatenteSelecionado?.lado === 'JOGADOR' && !combatenteSelecionado.agiu"
        label="Atacar"
        @click="modoAtacar = true"
      />
      <Button
        v-if="combatenteSelecionado?.lado === 'JOGADOR' && !combatenteSelecionado.agiu"
        label="Defender"
        @click="defender"
      />
      <Button label="Encerrar turno" @click="encerrarTurno" />
      <Button label="Render" severity="danger" @click="confirmarRender" />
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed, ref } from 'vue'
import type { BatalhaDto, CombatenteDto } from '../api/tipos'

const props = defineProps<{ batalha: BatalhaDto }>()
const emit = defineEmits<{
  acao: [{ tipo: string; combatenteId: string; x?: number; y?: number; alvoId?: string; turno: number }]
}>()

const selecionado = ref<{ x: number; y: number } | null>(null)
const modoMover = ref(false)
const modoAtacar = ref(false)

const todasAsPosicoes = computed(() => {
  const posicoes = []
  for (let y = 0; y < props.batalha.altura; y++) {
    for (let x = 0; x < props.batalha.largura; x++) {
      posicoes.push({ x, y })
    }
  }
  return posicoes
})

const combatenteSelecionado = computed(() => {
  if (!selecionado.value) return null
  return props.batalha.combatentes.find(c => c.x === selecionado.value?.x && c.y === selecionado.value?.y)
})

function isObstaculo(pos: { x: number; y: number }): boolean {
  return props.batalha.obstaculos.some(o => o.x === pos.x && o.y === pos.y)
}

function getCombatente(pos: { x: number; y: number }): CombatenteDto | undefined {
  return props.batalha.combatentes.find(c => c.x === pos.x && c.y === pos.y && c.vivo)
}

function isDestino(pos: { x: number; y: number }): boolean {
  return modoMover.value && selecionado.value?.x === pos.x && selecionado.value?.y === pos.y
}

function isAlvo(pos: { x: number; y: number }): boolean {
  return modoAtacar.value && selecionado.value?.x === pos.x && selecionado.value?.y === pos.y
}

function clicar(pos: { x: number; y: number }) {
  if (modoMover.value) {
    if (combatenteSelecionado.value) {
      emit('acao', {
        tipo: 'MOVER',
        combatenteId: combatenteSelecionado.value.id,
        x: pos.x,
        y: pos.y,
        turno: props.batalha.turno
      })
      modoMover.value = false
    }
  } else if (modoAtacar.value) {
    const alvo = getCombatente(pos)
    if (alvo) {
      emit('acao', {
        tipo: 'ATACAR',
        combatenteId: combatenteSelecionado.value!.id,
        alvoId: alvo.id,
        turno: props.batalha.turno
      })
      modoAtacar.value = false
    }
  } else {
    selecionado.value = pos
  }
}

function defender() {
  if (combatenteSelecionado.value) {
    emit('acao', {
      tipo: 'DEFENDER',
      combatenteId: combatenteSelecionado.value.id,
      turno: props.batalha.turno
    })
  }
}

function encerrarTurno() {
  emit('acao', {
    tipo: 'ENCERRAR_TURNO',
    combatenteId: 'dummy',
    turno: props.batalha.turno
  })
}

function confirmarRender() {
  if (confirm('Render encerra a batalha com derrota. Tem certeza?')) {
    emit('acao', {
      tipo: 'RENDER',
      combatenteId: 'dummy',
      turno: props.batalha.turno
    })
  }
}
</script>
```

### `MasmorrasView.vue` — seleção de nível e unidades
```vue
<template>
  <div class="masmorras-view">
    <h1>Masmorras</h1>
    
    <div v-if="batalhaAtiva">
      <Card class="retomar">
        <p>Você tem uma batalha em andamento!</p>
        <Button label="Continuar" @click="$router.push(`/batalhas/${batalhaAtiva.id}`)" />
      </Card>
    </div>
    
    <div class="niveis">
      <Card v-for="nivel in 5" :key="nivel" class="cartao-nivel">
        <template #title>Masmorra {{ nivel }}</template>
        
        <p v-if="nivel > masmorraNivelLiberado">Bloqueado</p>
        <div v-else>
          <p>Inimigos: {{ composicao[nivel]?.join(', ') }}</p>
          
          <MultiSelect
            v-model="unidadesEscolhidas"
            :options="unidadesDisponiveis"
            optionLabel="descricao"
            optionValue="id"
            :max-selected-labels="4"
          />
          
          <Button
            label="Entrar"
            @click="entrar(nivel)"
            :disabled="unidadesEscolhidas.length === 0 || unidadesEscolhidas.length > 4"
          />
        </div>
      </Card>
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed, ref, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { useVila } from '../composables/useVila'
import { jogoAPI } from '../api/jogo'

const router = useRouter()
const { vila, catalogo, carregar } = useVila()
const unidadesEscolhidas = ref<number[]>([])

const masmorraNivelLiberado = computed(() => vila.value?.masmorraNivelLiberado ?? 1)
const batalhaAtiva = computed(() => vila.value?.batalhaAtivaId)

const composicao = computed(() => {
  const comp: Record<number, string[]> = {}
  catalogo.value?.masmorras.forEach(m => {
    comp[m.nivel] = m.composicao
  })
  return comp
})

const unidadesDisponiveis = computed(() => {
  return (vila.value?.unidades ?? []).filter(u => u.status === 'DISPONIVEL').map(u => ({
    id: u.id,
    descricao: `${u.tipo} HP${u.hp}`
  }))
})

async function entrar(nivel: number) {
  try {
    const batalha = await jogoAPI.iniciarBatalha(nivel, unidadesEscolhidas.value)
    router.push(`/batalhas/${batalha.id}`)
  } catch (e: any) {
    console.error('Erro ao iniciar batalha:', e.mensagem)
  }
}

onMounted(() => carregar())
</script>

}

  margin-bottom: 24px;
}
</style>
```

### `BatalhaView.vue` — tela de combate tático
```vue
<template>
  <div v-if="batalha" class="batalha-view">
    <h1>Masmorra {{ batalha.masmorraNivel }} — Turno {{ batalha.turno }}</h1>
    
    <div class="conteudo">
      <GradeBatalha :batalha="batalha" @acao="agir" />
      
      <div class="direita">
        <Card class="log">
          <template #title>Log</template>
          <ScrollPanel style="width: 100%; height: 300px">
            <div v-for="(linha, idx) in batalha.log" :key="idx" class="log-linha">
              {{ linha }}
            </div>
          </ScrollPanel>
        </Card>
      </div>
    </div>
    
    <Dialog v-model:visible="mostrarResultado" header="Resultado da batalha">
      <div v-if="batalha.status === 'VITORIA'">
        <p class="sucesso">VITÓRIA!</p>
        <div v-if="batalha.loot">
          <h4>Loot</h4>
          <p v-for="[recurso, valor] in Object.entries(batalha.loot.recursos)" :key="recurso">
            {{ recurso }}: +{{ valor }}
          </p>
          <p v-for="[cultivo, qtd] in Object.entries(batalha.loot.sementes)" :key="cultivo">
            Semente {{ cultivo }}: +{{ qtd }}
          </p>
          <p v-for="item in batalha.loot.itens" :key="item.id">
            {{ item.modelo }} N{{ item.nivel }}
          </p>
        </div>
      </div>
      <div v-else-if="batalha.status === 'DERROTA'">
        <p class="derrota">DERROTA</p>
      </div>
      <Button label="Voltar" @click="$router.push('/masmorras')" />
    </Dialog>
  </div>
  <div v-else>
    Carregando...
  </div>
</template>

<script setup lang="ts">
import { computed, ref, onMounted, watch } from 'vue'
import { useRoute, useRouter } from 'vue-router'
import { jogoAPI } from '../api/jogo'
import { useVila } from '../composables/useVila'
import GradeBatalha from '../components/GradeBatalha.vue'
import type { BatalhaDto } from '../api/tipos'

const route = useRoute()
const router = useRouter()
const { carregar } = useVila()

const batalha = ref<BatalhaDto | null>(null)
const mostrarResultado = ref(false)

const batalhaId = computed(() => parseInt(route.params.id as string))

async function recarregarBatalha() {
  try {
    batalha.value = await jogoAPI.consultarBatalha(batalhaId.value)
    if (batalha.value.status !== 'EM_ANDAMENTO') {
      mostrarResultado.value = true
    }
  } catch (e) {
    console.error('Erro ao carregar batalha:', e)
  }
}

async function agir(acao: any) {
  try {
    batalha.value = await jogoAPI.agirBatalha(batalhaId.value, acao)
    if (batalha.value.status !== 'EM_ANDAMENTO') {
      mostrarResultado.value = true
      await carregar()
    }
  } catch (e: any) {
    console.error('Erro ao executar ação:', e.mensagem)
    // Recarregar em caso de conflito de turno
    if (e.status === 409) {
      await recarregarBatalha()
    }
  }
}

onMounted(() => recarregarBatalha())
</script>
```

## Critérios de aceite
- `npm run build` sem erros.
- MasmorrasView exibe 5 níveis com bloqueio, composição, seleção até 4 unidades, botão Entrar.
- Clique em Entrar chama API e navega a `/batalhas/{id}`.
- BatalhaView carrega batalha, exibe grade 8×8 com obstáculos/combatentes.
- Clique em combatente seleciona; Mover/Atacar/Defender/Encerrar turno/Render funcionam.
- Log exibe eventos.
- Status VITORIA/DERROTA mostra diálogo com loot (se vitória).
- Erro 409 turno desatualizado recarrega batalha.
