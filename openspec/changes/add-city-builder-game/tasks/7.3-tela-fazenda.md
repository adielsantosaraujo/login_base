## Notas de implementação

### `FazendaView.vue` — tela com DataTable e estoque
```vue
<template>
  <div class="fazenda-view">
    <h1>Fazenda ({{ nivelFazenda }})</h1>
    
    <h2>Canteiros</h2>
    <DataTable :value="vila?.canteiros ?? []" responsiveLayout="scroll">
      <Column field="posicao" header="Posição" />
      <Column field="cultivo" header="Cultivo" />
      <Column field="comidaPorHora" header="Comida/h" />
      <template #body="{ data }">
        <tr>
          <td>{{ data.posicao }}</td>
          <td>
            <Select
              v-model="culturaEscolhido[data.posicao]"
              :options="cultivosDisponiveis"
              optionLabel="label"
              optionValue="value"
            />
          </td>
          <td>{{ data.comidaPorHora }}</td>
          <td>
            <Button
              icon="pi pi-check"
              @click="plantar(data.posicao)"
              :disabled="naoPodePlantar(data.posicao)"
              label="Plantar"
            />
          </td>
        </tr>
      </template>
    </DataTable>
    
    <h2>Estoque de Sementes</h2>
    <div class="estoque">
      <div v-for="[cultivo, qtd] in Object.entries(vila?.sementes ?? {})" :key="cultivo">
        <span>{{ cultivo }}: {{ qtd }}</span>
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed, ref, onMounted } from 'vue'
import { useVila } from '../composables/useVila'
import { jogoAPI } from '../api/jogo'
import { useToast } from 'primevue/usetoast'
import type { Cultivo } from '../api/tipos'

const { vila, carregar, iniciarPoll } = useVila()
const toast = useToast()
const culturaEscolhido = ref<Record<number, string>>({})

const nivelFazenda = computed(() => {
  return vila.value?.predios.find(p => p.tipo === 'FAZENDA')?.nivel ?? 0
})

const cultivosDisponiveis = computed(() => {
  // TRIGO sempre disponível, demais dependem do nível da masmorra
  return [
    { label: 'TRIGO', value: 'TRIGO' },
    { label: 'MILHO', value: 'MILHO' },
    { label: 'BATATA', value: 'BATATA' },
    { label: 'ABOBORA_DOURADA', value: 'ABOBORA_DOURADA' },
  ]
})

function naoPodePlantar(posicao: number): boolean {
  const cultivo = culturaEscolhido.value[posicao]
  if (!cultivo || cultivo === 'TRIGO') return false
  const sementes = vila.value?.sementes[cultivo as Cultivo] ?? 0
  return sementes <= 0
}

async function plantar(posicao: number) {
  const cultivo = culturaEscolhido.value[posicao]
  if (!cultivo) return
  
  try {
    await jogoAPI.plantar(posicao, cultivo)
    await carregar()
  } catch (e: any) {
    toast.add({ severity: 'error', detail: e.mensagem, life: 3000 })
  }
}

onMounted(() => {
  carregar()
  iniciarPoll()
})
</script>

<style scoped>
.fazenda-view {
  padding: 16px;
}

.estoque {
  display: grid;
  grid-template-columns: repeat(auto-fit, minmax(150px, 1fr));
  gap: 8px;
  padding: 16px;
  background: var(--surface);
}
</style>
```

## Critérios de aceite
- `npm run build` sem erros.
- FazendaView renderiza DataTable com canteiros.
- Select de cultivo funciona; plantio com TRIGO não exige semente.
- Botão Plantar desabilitado se sem semente.
- Estoque de sementes exibe valores.
- Erros 422 exibem Toast.
